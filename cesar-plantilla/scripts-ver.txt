<script>
        // 1. Configuración de URLs
        const APPS_SCRIPT_URL_LEAD = "https://script.google.com/macros/s/AKfycbxncWtl2du2-DRDmP52p5y8gKJG2DmcBRRe3OIqKHsqGb_qvBBY5uL_-HvIAGm3WxEJoQ/exec";
        const APPS_SCRIPT_URL_INHOUSE = "https://script.google.com/macros/s/AKfycbzOir6cJzWXuc-nX7B23qZ9N4eGN-eyTy1a17Bh2tteFhhLLoiCASGk9zYLkoOQBVzTQA/exec";

        // 2. Única función de envío (Actualizada para abrir Modal de Gracias)
        async function sendToApps(paramsObj, targetUrl) {
            const params = new URLSearchParams(paramsObj);
            try {
                Swal.fire({
                    title: 'Enviando...',
                    text: 'Por favor espere',
                    allowOutsideClick: false,
                    didOpen: () => Swal.showLoading()
                });

                // Lanza el fetch sin esperar (queda corriendo en background)
                fetch(targetUrl, {
                    method: 'POST',
                    mode: 'no-cors',
                    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
                    body: params.toString()
                });

                // Espera 3 segundos fijos
                await new Promise(resolve => setTimeout(resolve, 3000));

                // Cerrar Swal y abrir el modal personalizado de gracias
                Swal.close();
                const modalGraciasEl = document.getElementById('modalGracias');
                if (modalGraciasEl && typeof bootstrap !== 'undefined') {
                    const modalGracias = bootstrap.Modal.getInstance(modalGraciasEl) || new bootstrap.Modal(modalGraciasEl);
                    modalGracias.show();
                }
                return true;

            } catch (err) {
                console.error('Error enviando a Apps Script:', err);
                Swal.fire({
                    icon: 'error',
                    title: 'Error al enviar',
                    text: 'Inténtalo nuevamente más tarde.',
                    confirmButtonColor: '#dc3545'
                });
                return false;
            }
        }

        // 3. Manejador de Leads (3 campos)

        /* USAR ESTAS RUTA DE DESTINO
        const cursosImagen = [
  "CURSO: SIAF WEB - JUNIO",
  "CURSO: SEACE Y PLADICOP - JUNIO",
  "CURSO: PLANEAMIENTO ESTRATÉGICO - MAYO",
  "CURSO: OFIMATICA - MAYO",
  "CURSO: SIAF SIGA SEACE Y PLADICOP - MAYO",
  "DIPLOMADO SIAF, SIGA Y SEACE Y PLADICOP - MAYO",
  "CONTRATACIONES + LICITACIONES - MAYO (YAJA)",
  "CONTRATACIONES + LICITACIONES - MAYO (ROMI)",
  "Certificación OECE - MAYO",
  "SIGA MEF MAYO",
  "DIPLOMADO CONTRATACIONES MAYO"
  "SOPORTE TÉCNICO SIAF Y SIGA 2026 - JUNIO"
];
        */

        function handleLead(e) {
            e.preventDefault();
            const form = e.target;
            const payload = {
                origen: 'Formulario Web',
                nombreCompleto: form.querySelector('input[name="nombre"]')?.value || '',
                correo: form.querySelector('input[name="correo"]')?.value || '',
                celular: form.querySelector('input[name="celular"]')?.value || '',

                // VARIABLE 1: Para identificar la PÁGINA (Hoja) en el Excel
                nombreHojaDestino: "SOPORTE TÉCNICO SIAF Y SIGA 2026 - JUNIO",

                // VARIABLE 2: El texto real que quieres que aparezca en la Columna G
                cursoNombre: "Curso en vivo de Soporte Técnico y Soluciones Informáticas en el SIAF y SIGA 2026"
            };

            sendToApps(payload, APPS_SCRIPT_URL_LEAD).then(ok => {
                if (ok) form.reset();
            });
            return false;
        }

        // 4. Lógica del DOM y Formularios Inhouse
        document.addEventListener('DOMContentLoaded', function () {
            const inhouseForm = document.getElementById('inhouseForm');

            if (inhouseForm) {
                inhouseForm.addEventListener('submit', function (ev) {
                    ev.preventDefault();

                    const payload = {
                        origen: 'FORMULARIO WEB',
                        nombres: inhouseForm.querySelector('input[name="nombres"]')?.value || '',
                        correo: inhouseForm.querySelector('input[name="correo"]')?.value || '',
                        telefono: inhouseForm.querySelector('input[name="telefono"]')?.value || '',
                        institucion: inhouseForm.querySelector('input[name="institucion"]')?.value || '',
                        cantidadAlumnos: inhouseForm.querySelector('select[name="cantidadAlumnos"]')?.value || '',
                        nivelCurso: inhouseForm.querySelector('select[name="nivelCurso"]')?.value || '',
                        // Nombre del Curso o Diplomado
                        requerimiento: 'Curso en vivo de Soporte Técnico y Soluciones Informáticas en el SIAF y SIGA 2026',
                        urlWha: window.location.href
                    };

                    sendToApps(payload, APPS_SCRIPT_URL_INHOUSE).then(ok => {
                        if (ok) {
                            // Primero cerramos el modal del formulario si existe
                            try {
                                const modalEl = document.getElementById('inhouseModal');
                                if (modalEl && typeof bootstrap !== 'undefined') {
                                    const modal = bootstrap.Modal.getInstance(modalEl) || new bootstrap.Modal(modalEl);
                                    modal.hide();
                                }
                            } catch (err) {
                                console.warn('Error al cerrar modal inhouse', err);
                            }
                            inhouseForm.reset();
                        }
                    });
                });
            }

            // --- Lógica Visual de UI ---
            const inhouseModal = document.getElementById('inhouseModal');
            if (inhouseModal) {
                inhouseModal.addEventListener('show.bs.modal', function () {
                    document.querySelector('.inhouse-section').style.display = 'none';
                    if (window.innerWidth < 768) {
                        document.querySelector('.c-design-column').style.display = 'none';
                        var btnClose = document.createElement('button');
                        btnClose.type = 'button';
                        btnClose.className = 'btn-close position-absolute';
                        btnClose.setAttribute('data-bs-dismiss', 'modal');
                        btnClose.style.top = '10px'; btnClose.style.right = '10px'; btnClose.style.zIndex = '1000';
                        document.querySelector('.c-form-column').insertBefore(btnClose, document.querySelector('.c-form-column').firstChild);
                    }
                });

                inhouseModal.addEventListener('hidden.bs.modal', function () {
                    document.querySelector('.inhouse-section').style.display = 'block';
                    document.querySelector('.c-design-column').style.display = 'flex';
                    var btnClose = document.querySelector('.c-form-column .btn-close');
                    if (btnClose) btnClose.remove();
                });
            }

            // Ocultar panel desktop al final del scroll
            const panel = document.getElementById('foot');
            if (panel && window.innerWidth >= 992) {
                panel.style.transition = 'opacity .3s ease';
                window.addEventListener('scroll', function () {
                    const rect = panel.getBoundingClientRect();
                    if (rect.bottom < 0) { panel.style.opacity = '0'; panel.style.pointerEvents = 'none'; }
                    else { panel.style.opacity = '1'; panel.style.pointerEvents = 'auto'; }
                });
            }
        });

        // 5. Inicialización del Carrusel
        $(document).ready(function () {
            $('.carrusel-testimonios').slick({
                dots: true,
                infinite: true,
                speed: 800,
                slidesToShow: 3,
                slidesToScroll: 1,
                autoplay: true,
                autoplaySpeed: 3000,
                arrows: true,
                responsive: [
                    { breakpoint: 1200, settings: { slidesToShow: 2 } },
                    { breakpoint: 768, settings: { slidesToShow: 1, arrows: false } }
                ]
            });
        });
    </script>
